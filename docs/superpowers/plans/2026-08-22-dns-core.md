> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Close all IPv6 and DoH leakage vectors, add an in-memory DNS cache, implement race-forwarding for upstream resilience, and guarantee safe teardown of system DNS settings.

**Architecture:** Modifies the DNS resolver to be type-aware (synthesizing AAAA, NOERROR for HTTPS). Adds a concurrent LRU cache for upstream responses. Replaces sequential upstream querying with a race-forward model. Upgrades the Swift MenuBar app to enumerate all network interfaces instead of just the default route, and adds robust signal handling to the Go daemon for safe teardown.

**Tech Stack:** Go 1.22+, `github.com/miekg/dns`, Swift 6, macOS `networksetup`

## Global Constraints
- Platform: macOS 15+
- DNS cache RAM budget: < 8 MB (bounded LRU)
- Cache lookup must return in < 0.2 ms
- Upstream race timeout: 500 ms before fallback
- Interface hooking applies to all services returned by `networksetup -listallnetworkservices`

```go
func sendBlockedResponse(w dns.ResponseWriter, r *dns.Msg) {
	m := new(dns.Msg)
	m.SetReply(r)
	m.Authoritative = true

	for _, q := range r.Question {
		switch q.Qtype {
		case dns.TypeA:
			rr, _ := dns.NewRR(q.Name + " 3600 IN A 0.0.0.0")
			m.Answer = append(m.Answer, rr)
		case dns.TypeAAAA:
			rr, _ := dns.NewRR(q.Name + " 3600 IN AAAA ::")
			m.Answer = append(m.Answer, rr)
		case 65, 64: // HTTPS, SVCB
			// NOERROR, empty answer section
		case dns.TypeANY:
			rrA, _ := dns.NewRR(q.Name + " 3600 IN A 0.0.0.0")
			rrAAAA, _ := dns.NewRR(q.Name + " 3600 IN AAAA ::")
			m.Answer = append(m.Answer, rrA, rrAAAA)
		default:
			// For simplicity on other types for a blocked domain, empty NOERROR is standard
		}
	}
	w.WriteMsg(m)
}
```

```go
package dnsd

import (
	"sync"
	"time"
	"github.com/miekg/dns"
)

type cacheEntry struct {
	msg       *dns.Msg
	expiresAt time.Time
	lastUsed  time.Time
}

type DNSCache struct {
	mu         sync.RWMutex
	entries    map[string]*cacheEntry
	maxEntries int
}

func NewDNSCache(max int) *DNSCache {
	return &DNSCache{
		entries:    make(map[string]*cacheEntry),
		maxEntries: max,
	}
}

func cacheKey(qname string, qtype uint16) string {
	// Simple string concat for key. In production, consider a struct key.
	return qname + "|" + string(rune(qtype))
}

func (c *DNSCache) Get(qname string, qtype uint16) (*dns.Msg, bool) {
	c.mu.Lock()
	defer c.mu.Unlock()
	
	key := cacheKey(qname, qtype)
	entry, ok := c.entries[key]
	if !ok {
		return nil, false
	}
	
	if time.Now().After(entry.expiresAt) {
		delete(c.entries, key)
		return nil, false
	}
	
	entry.lastUsed = time.Now()
	
	// Decrement TTL in the returned message (deep copy is required in production, 
	// for now we'll return a copy of the msg with adjusted TTLs)
	msgCopy := entry.msg.Copy()
	ttlRemaining := uint32(time.Until(entry.expiresAt).Seconds())
	for _, a := range msgCopy.Answer {
		a.Header().Ttl = ttlRemaining
	}
	
	return msgCopy, true
}

func (c *DNSCache) Set(qname string, qtype uint16, msg *dns.Msg) {
	if len(msg.Answer) == 0 {
		return
	}
	
	c.mu.Lock()
	defer c.mu.Unlock()
	
	if len(c.entries) >= c.maxEntries {
		c.evictOldest()
	}
	
	ttl := msg.Answer[0].Header().Ttl
	
	key := cacheKey(qname, qtype)
	c.entries[key] = &cacheEntry{
		msg:       msg.Copy(),
		expiresAt: time.Now().Add(time.Duration(ttl) * time.Second),
		lastUsed:  time.Now(),
	}
}

func (c *DNSCache) evictOldest() {
	var oldestKey string
	var oldestTime time.Time
	
	first := true
	for k, v := range c.entries {
		if first || v.lastUsed.Before(oldestTime) {
			oldestTime = v.lastUsed
			oldestKey = k
			first = false
		}
	}
	if !first {
		delete(c.entries, oldestKey)
	}
}

func (c *DNSCache) Flush() {
	c.mu.Lock()
	c.entries = make(map[string]*cacheEntry)
	c.mu.Unlock()
}
```

```go
package dnsd

import (
	"context"
	"errors"
	"time"
	"github.com/miekg/dns"
)

func RaceForward(r *dns.Msg, upstreams []string, timeout time.Duration) (*dns.Msg, error) {
	if len(upstreams) == 0 {
		return nil, errors.New("no upstreams configured")
	}

	ctx, cancel := context.WithTimeout(context.Background(), timeout)
	defer cancel()

	resultCh := make(chan *dns.Msg, len(upstreams))
	
	for _, upstream := range upstreams {
		go func(server string) {
			client := &dns.Client{Timeout: timeout}
			resp, _, err := client.ExchangeContext(ctx, r.Copy(), server)
			if err == nil && resp != nil {
				select {
				case resultCh <- resp:
				default:
				}
			}
		}(upstream)
	}

	select {
	case resp := <-resultCh:
		return resp, nil
	case <-ctx.Done():
		return nil, errors.New("upstream timeout or all failed")
	}
}
```
